async function adminExecuteMoneyTransfer(action) { if(currentUser.toLowerCase() !== 'qubax') { showToast("Brak uprawnień administratora!", "error"); return; } const amt = parseInt(document.getElementById("admin-amount").value, 10); const target = document.getElementById("admin-target").value; if (isNaN(amt) || amt <= 0) { showToast("Podaj poprawną kwotę!", "error"); return; } if (target === "self") { if(action === 'add') { game.money += amt; } else { game.money = Math.max(0, game.money - amt); } updateBalance(false); updateMapStats(); // Natychmiast wymuś zapis nowego (zmniejszonego lub zwiększonego) salda do chmury await syncGameToCloud(true); showToast("Zaktualizowano własne saldo i zapisano w sieci."); } else { try { const rows = await sbRequest(`companies?username=eq.${encodeURIComponent(target)}&select=money`); if (!rows || rows.length === 0) { showToast("Nie znaleziono firmy w bazie.", "error"); return; } const currentMoney = Number(rows[0].money || 0); const updatedMoney = action === 'add' ? currentMoney + amt : Math.max(0, currentMoney - amt); await sbRequest(`companies?username=eq.${encodeURIComponent(target)}`, { method: "PATCH", body: JSON.stringify({ money: updatedMoney }) }); showToast(`Zaktualizowano saldo gracza ${target}: ${updatedMoney.toLocaleString()} PLN`); } catch (e) { showToast("Błąd aktualizacji salda w bazie.", "error"); } } } // 2. Poprawiona synchronizacja z chmurą (wyłączona fałszywa detekcja dopłat przy odejmowaniu) async function syncGameToCloud(force = false) { if (!currentUser || isCloudSyncing) return; const now = Date.now(); if (!force && now - lastCloudSyncTime < 4000) return; isCloudSyncing = true; lastCloudSyncTime = now; setCloudSyncStatus("syncing"); try { // Synchronizujemy stan bez automatycznego dopisywania różnic, jeśli to my sami zmieniliśmy saldo const payload = { username: currentUser, company_name: game.companyName || currentUser, profile: game.profile || "urban", money: Math.floor(game.money || 0), vehicles_count: game.vehicles ? game.vehicles.length : 0, lines_count: game.lines ? game.lines.length : 0, state_json: game, updated_at: new Date().toISOString() }; await sbRequest("companies?on_conflict=username", { method: "POST", headers: { "Prefer": "resolution=merge-duplicates" }, body: JSON.stringify(payload) }); setCloudSyncStatus("ok"); } catch (e) { console.warn("Błąd zapisu do Supabase:", e); setCloudSyncStatus("err"); } finally { isCloudSyncing = false; } }
